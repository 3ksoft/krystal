 93	//   lossRows[m] = logZ - logits[m, targets[u1 + m]]
 94	//   dLogits[m,v] = (exp(logits[m,v] - logZ) - one_hot(targets[u1 + m], v)) / M
 95	//
 96	// OpParams:
 97	//   tokenCount = M, outputDim = V
 98	//   inputOffset  = logits  [M,V]
 99	//   outputOffset = dLogits [M,V]
100	//   auxOffset    = lossRows [M]
101	//   u1 = target-id base offset inside the targets buffer
102	//
103	// A target outside [0, V) writes NaN to its loss row so any test that passes
104	// one fails loudly; the host validates targets before dispatch.
105
106	  let m = wid.x;
107	  if (m >= op.tokenCount) { return; }
108	  let V = op.outputDim;
109	  let base = op.inputOffset + m * V;
110	  let targetId = targets[op.u1 + m];
111
112	  // Pass 1: row max.
113	  var localMax = -1.0e30;
114	  var v = lid.x;
115	  loop {
116	    if (v >= V) { break; }
117	    localMax = max(localMax, arena[base + v]);
118	    v += WG;
119	  }
120	  reduceF32[lid.x] = localMax;
121	  workgroupBarrier();
122	  var width = WG >> 1u;
123	  loop {
124	    if (width == 0u) { break; }
125	    if (lid.x < width) { reduceF32[lid.x] = max(reduceF32[lid.x], reduceF32[lid.x + width]); }
126	    workgroupBarrier();
127	    width >>= 1u;
128	  }
129	  let rowMax = reduceF32[0];
130	  workgroupBarrier();
131
132	  // Pass 2: sum of exp(logit - rowMax).
133	  var localSum = 0.0;
134	  v = lid.x;
135	  loop {
136	    if (v >= V) { break; }
137	    localSum += exp(arena[base + v] - rowMax);
138	    v += WG;
139	  }
140	  reduceF32[lid.x] = localSum;
141	  workgroupBarrier();
142	  width = WG >> 1u;
143	  loop {
144	    if (width == 0u) { break; }
145	    if (lid.x < width) { reduceF32[lid.x] += reduceF32[lid.x + width]; }
146	    workgroupBarrier();
147	    width >>= 1u;
148	  }
149	  let logZ = rowMax + log(reduceF32[0]);
150	  workgroupBarrier();
151
152	  // Pass 3: dLogits for the owned slice + the row's loss.
153	  if (lid.x == 0u) {
154	    if (targetId < V) {
155	      arena[op.auxOffset + m] = logZ - arena[base + targetId];
156	    } else {
157	      // NaN via runtime division (a literal NaN is rejected at parse time).
158	      let zero = 0.0;
159	      arena[op.auxOffset + m] = zero / zero;
160	    }
161	  }
162	  v = lid.x;
163	  loop {
164	    if (v >= V) { break; }
165	    let prob = exp(arena[base + v] - logZ);
166	    let oneHot = select(0.0, 1.0, v == targetId);
167	    arena[op.outputOffset + m * V + v] = (prob - oneHot) / f32(op.tokenCount);
168	    v += WG;
169	  }

173	      value += attentionScores[k] * arena[op.aux2Offset + k * H + dim];
174	    }

 74//                                   backward, from the recomputed probs
 75var<workgroup> poolKeyScores: array<f32, 8>;
 76var<workgroup> poolValueScores: array<f32, 8>;
 77var<workgroup> poolKeyGrads: array<f32, 8>;
 78var<workgroup> poolValueGrads: array<f32, 8>;

 83@workgroup_size(64, 1, 1)

 85fn krystal_pool_backward(
 86	@builtin(local_invocation_id) lid: vec3u,
 87	@builtin(workgroup_id) wid: vec3u
 88) {
 89
 90	// Body-only compute shader source.
 91	// Sandblaster owns @compute, @workgroup_size and the entry-point signature.
 92	// entryPoint: krystal_pool_backward
 93	// workgroupSize: [64, 1, 1]
 94	// builtins: workgroup_id -> wid, local_invocation_id -> lid
 95	//
 96	// Record key/value pooling backward (M3, §17 item 7 — record mixer training
 97	// path). Mirrors krystal_pool forward:
 98	//
 99	//   key[r,h]   = sum_j pk[j] * s[j,h],   pk = softmax(qk . s_j / sqrt(H))
100	//   value[r,h] = sum_j pv[j] * s[j,h],   pv = softmax(qv . s_j / sqrt(H))
101	//
102	// Given upstream dKeys/dValues, compute:
103	//   dPk[j]          = dot(dKeys[r], s_j)      (through the weighted sum)
104	//   dScoreK[j]      = pk[j] * (dPk[j] - rowSumK), rowSumK = sum pk*dPk
105	//   dQk[d]          = scale * sum_j dScoreK[j] * s[j,d]
106	//   dS[j,d]        += pk[j]*dKeys[r,d] + scale*dScoreK[j]*qk[d]
107	//   (same for the value side with qv/pv/dValues)
108	//
109	// The fieldStates gradient contributions are disjoint per record (each record
110	// owns its compact token range), so this shader writes dFieldStates directly.
111	// The pool-query gradient dPool accumulates across records, so each workgroup
112	// writes its per-record partial into dPoolPartial [R, 2, H]; the
113	// krystal_pool_dpool pass reduces those into the final dPool [2, H].
114	// Recomputed softmaxes are deterministic (same math as the forward).
115	//
116	// OpParams:
117	//   tokenCount = recordCount, inputDim = H
118	//   inputOffset  = fieldStates [T_active, H]
119	//   auxOffset    = recordIndices [recordCount] (u32 payloads)
120	//   aux2Offset   = recordCompactOffset [maxRecords] (u32)
121	//   aux3Offset   = recordCompactCount [maxRecords] (u32)
122	//   aux4Offset   = dKeys   [recordCount, H]
123	//   aux5Offset   = dValues [recordCount, H]
124	//   outputOffset = dFieldStates [T_active, H]
125	//   aux6Offset   = dPoolPartial [recordCount, 2, H]  (key row 0, value row 1)
126	//   weight32     = pool queries [2, H] (qk row 0, qv row 1)
127	//   pool-backward-scores include: workgroup arrays (8 slots each)
128
129	  let r = wid.x;
130	  let recordCount = op.tokenCount;
131	  let H = op.inputDim;
132	  if (r >= recordCount) { return; }
133
134	  let slot = bitcast<u32>(arena[op.auxOffset + r]);
135	  let start = bitcast<u32>(arena[op.aux2Offset + slot]);
136	  let count = bitcast<u32>(arena[op.aux3Offset + slot]);
137
138	  let scale = 1.0 / sqrt(f32(H));
139
140	  // Pass 1: scores + dPk/dPv per active token (one lane per token).
141	  var j = lid.x;
142	  loop {
143	    if (j >= count) { break; }
144	    var ks = 0.0;
145	    var vs = 0.0;
146	    var dPk = 0.0;
147	    var dPv = 0.0;
148	    let base = op.inputOffset + (start + j) * H;
149	    let dkBase = op.aux4Offset + r * H;
150	    let dvBase = op.aux5Offset + r * H;
151	    for (var d = 0u; d < H; d++) {
152	      let s = arena[base + d];
153	      ks += weight32[d] * s;
154	      vs += weight32[H + d] * s;
155	      dPk += arena[dkBase + d] * s;
156	      dPv += arena[dvBase + d] * s;
157	    }
158	    poolKeyScores[j] = ks * scale;
159	    poolValueScores[j] = vs * scale;
160	    poolKeyGrads[j] = dPk;
161	    poolValueGrads[j] = dPv;
162	    j += WG;
163	  }
164	  workgroupBarrier();
165
166	  // Pass 2: independent softmax for key/value rows (single lane), then the
167	  // softmax backward: dScore = p * (dP - rowSum).
168	  if (lid.x == 0u) {
169	    var kMax = -3.402823466e+38;
170	    var vMax = -3.402823466e+38;
171	    for (var k = 0u; k < count; k++) {
172	      kMax = max(kMax, poolKeyScores[k]);
173	      vMax = max(vMax, poolValueScores[k]);
174	    }
175	    var kSum = 0.0;
176	    var vSum = 0.0;
177	    for (var k = 0u; k < count; k++) {
178	      let ke = exp(poolKeyScores[k] - kMax);
179	      let ve = exp(poolValueScores[k] - vMax);
180	      poolKeyScores[k] = ke;
181	      poolValueScores[k] = ve;
182	      kSum += ke;
183	      vSum += ve;
184	    }
185	    let kInv = 1.0 / max(kSum, 1e-20);
186	    let vInv = 1.0 / max(vSum, 1e-20);
187	    var kRowSum = 0.0;
188	    var vRowSum = 0.0;
189	    for (var k = 0u; k < count; k++) {
190	      poolKeyScores[k] *= kInv;
191	      poolValueScores[k] *= vInv;
192	      kRowSum += poolKeyScores[k] * poolKeyGrads[k];
193	      vRowSum += poolValueScores[k] * poolValueGrads[k];
194	    }
195	    for (var k = 0u; k < count; k++) {
196	      poolKeyGrads[k] = poolKeyScores[k] * (poolKeyGrads[k] - kRowSum);
197	      poolValueGrads[k] = poolValueScores[k] * (poolValueGrads[k] - vRowSum);
198	    }
199	  }
200	  workgroupBarrier();
201
202	  // Pass 3: fieldStates gradient (disjoint per record) + per-record dPool
203	  // partials (lanes over dims).
204	  var dim = lid.x;
205	  loop {
206	    if (dim >= H) { break; }
207	    var dQk = 0.0;
208	    var dQv = 0.0;
209	    for (var k = 0u; k < count; k++) {
210	      let base = op.inputOffset + (start + k) * H + dim;
211	      let s = arena[base];
212	      dQk += poolKeyGrads[k] * s;
213	      dQv += poolValueGrads[k] * s;
214	      let dS = poolKeyScores[k] * arena[op.aux4Offset + r * H + dim] +
215	        poolValueScores[k] * arena[op.aux5Offset + r * H + dim] +
216	        scale * (poolKeyGrads[k] * weight32[dim] + poolValueGrads[k] * weight32[H + dim]);
217	      arena[op.outputOffset + (start + k) * H + dim] += dS;
218	    }
219	    arena[op.aux6Offset + r * 2 * H + dim] = dQk * scale;
220	    arena[op.aux6Offset + r * 2 * H + H + dim] = dQv * scale;
221	    dim += WG;
222	  }

225}

 96	//   aux3Offset   = p       [Q, R]
 97	//   aux4Offset   = dGather [Q, H]
 98	//   outputOffset = dQProj  [Q, H]
 99	//   aux5Offset   = dKProj  [R, H]
100	//   aux6Offset   = dValue  [R, H]
101	//
102	// OpParams extras:
103	//   u0 = R
104	//   u1 = gold offset in arena f32 elements ([Q] u32 payloads; only read
105	//        when u2 != 0)
106	//   u2 = skipInvalidRows (0/1): when set, query rows with an INVALID
107	//        pointer target are skipped in the dValue sum (FOLLOW_UP2 Fix A).
108	//        dScore rows are already zeroed for those rows by the companion
109	//        scores pass, so dQProj/dKProj need no guard; the p * dGather term
110	//        in dValue must be skipped explicitly, otherwise the degenerate
111	//        all-masked uniform p of an arity-0 / unlabelled row leaks onto
112	//        distractor records and into the shared selector parameters.
113
114	  let Q = op.tokenCount;
115	  let R = op.u0;
116	  let H = op.inputDim;
117	  let qTotal = Q * H;
118	  let rTotal = R * H;
119	  let linear = gid.x;
120	  if (linear >= qTotal + 2u * rTotal) { return; }
121
122	  let scale = 1.0 / sqrt(f32(H));
123	  let col = linear % H;
124
125	  if (linear < qTotal) {
126	    // dQProj[i, col] = scale * sum_j dScore[i,j] * kProj[j, col]
127	    let i = linear / H;
128	    var sum = 0.0;
129	    for (var j = 0u; j < R; j++) {
130	      sum += arena[op.inputOffset + i * R + j] * arena[op.aux2Offset + j * H + col];
131	    }
132	    arena[op.outputOffset + i * H + col] = sum * scale;
133	    return;
134	  }
135	  if (linear < qTotal + rTotal) {
136	    // dKProj[j, col] = scale * sum_i dScore[i,j] * qProj[i, col]
137	    let j = (linear - qTotal) / H;
138	    var sum = 0.0;
139	    for (var i2 = 0u; i2 < Q; i2++) {
140	      sum += arena[op.inputOffset + i2 * R + j] * arena[op.auxOffset + i2 * H + col];
141	    }
142	    arena[op.aux5Offset + j * H + col] = sum * scale;
143	    return;
144	  }
145	  // dValue[j, col] = sum_i p[i,j] * dGather[i, col], skipping rows with an
146	  // INVALID pointer target when u2 != 0 (FOLLOW_UP2 Fix A).
147	  let j = (linear - qTotal - rTotal) / H;
148	  var sum = 0.0;
149	  for (var i2 = 0u; i2 < Q; i2++) {
150	    if (op.u2 != 0u) {
151	      let gold = bitcast<u32>(arena[op.u1 + i2]);
152	      if (gold == 0xffffffffu) { continue; }
153	    }
154	    sum += arena[op.aux3Offset + i2 * R + j] * arena[op.aux4Offset + i2 * H + col];
155	  }
156	  arena[op.aux6Offset + j * H + col] = sum;
157

 83	//
 84	//   logits[q,c] = sum_{d in [0,3H)} ctx[q,d] * Wh[c,d]

 86	// ctx[q] = concat(queryOutput[q], intentGather[q], argGather[q]) — read from

 92	//   dArgGather[q,d]    = sum_c dLogits[q,c] * Wh[c, 2H + d]
 93	//   dWh[c,d']          = sum_q dLogits[q,c] * ctx[q, d']   (d' in [0, 3H))

 95	// The linear gid is split into four blocks like the qkv backward:
 96	//   [0, QH)                    -> dQueryOutput
 97	//   [QH, 2QH)                  -> dIntentGather
 98	//   [2QH, 3QH)                 -> dArgGather
